> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Dashboard de cumplimiento ISO 27001 Anexo A (2022) por proyecto: estado de los 93 controles derivado on-read de señal real (findings/remediation) + attestation manual persistida, con API y evidence pack JSON con hash de custodia.

```ts
import { index, pgEnum, pgTable, text, timestamp, unique, uuid } from "drizzle-orm/pg-core";
import { projects, users } from "./index";

export const complianceAttestationStatusEnum = pgEnum("compliance_attestation_status", [
  "compliant",
  "non_compliant",
  "not_applicable",
]);

export const complianceAttestations = pgTable(
  "compliance_attestations",
  {
    id: uuid("id").primaryKey().defaultRandom(),
    projectId: uuid("project_id")
      .notNull()
      .references(() => projects.id, { onDelete: "cascade" }),
    frameworkId: text("framework_id").notNull().default("iso27001-annex-a"),
    controlCode: text("control_code").notNull(),
    status: complianceAttestationStatusEnum("status").notNull(),
    note: text("note"),
    evidenceUrl: text("evidence_url"),
    assertedBy: uuid("asserted_by").references(() => users.id, { onDelete: "set null" }),
    createdAt: timestamp("created_at", { withTimezone: true }).notNull().defaultNow(),
    updatedAt: timestamp("updated_at", { withTimezone: true }).notNull().defaultNow(),
  },
  (t) => [
    unique("compliance_attestations_project_framework_control_unique").on(
      t.projectId,
      t.frameworkId,
      t.controlCode
    ),
    index("idx_compliance_attestation_project").on(t.projectId),
  ]
);
```

```ts
import type { ControlDefinition, ThemeKind } from "./catalog/iso27001-annex-a";
import { ISO27001_ANNEX_A } from "./catalog/iso27001-annex-a";
import { verdictFromScore } from "@/server/reports/exec-summary";

export type ControlStatus =
  | "compliant" // confirmado por señal auto o attestation con señal positiva
  | "non_compliant" // findings abiertos mapeados (o attestation non_compliant)
  | "not_applicable" // solo attestation
  | "declared" // attestation compliant SIN señal auto
  | "no_signal"; // sin attestation ni señal → pendiente de declaración

export interface FindingLike {
  id: string;
  projectId: string;
  severity: string;
  status: string;
  title: string;
  evidence: unknown;
}

export interface AttestationDTO {
  controlCode: string;
  status: "compliant" | "non_compliant" | "not_applicable";
  note: string | null;
  evidenceUrl: string | null;
}

export interface EvaluationContext {
  openFindings: FindingLike[];
  verifiedRemediations: Array<{
    title: string | null;
    vulnerabilityTitle: string | null;
    assessmentId: string | null;
  }>;
}

type AutoSignal = { status: "compliant" | "non_compliant"; refs: string[] } | null;

// Reglas deterministas RESTRINGIDAS (anti sobre-afirmación; lección B3 de la auditoría).
// Solo matchean cuando la correspondencia es inequívoca. Cambiar aquí = actualizar engine.test.ts.
const RULES: Array<{ code: string; patterns: RegExp[] }> = [
  { code: "A.8.16", patterns: [/\b(si?e?m|monitore[oó]|monitoring|alertas)\b/i] },
  { code: "A.8.15", patterns: [/\b(log(ging)?|registro(s)? de auditor[ií]a)\b/i] },
  { code: "A.8.5", patterns: [/\b(autenticaci[oó]n|mfa|2fa|credencial(es)?)\b/i] },
  { code: "A.8.2", patterns: [/\b(privileg|admin(r)?(istrador)?|root|superuser)\b/i] },
  { code: "A.8.8", patterns: [/\b(vulnerab|cve|parche|patch)\b/i] },
  { code: "A.8.7", patterns: [/\b(malware|virus|troyano|ransomware)\b/i] },
  { code: "A.8.12", patterns: [/\b(fuga|leak|filtraci[oó]n)\b/i] },
  { code: "A.8.13", patterns: [/\b(backup|respaldo|copia de seguridad)\b/i] },
  { code: "A.8.24", patterns: [/\b(tls|ssl|cifra|cifrado|crypto)\b/i] },
  { code: "A.8.28", patterns: [/\b(xss|sql injection|rce|ССРF)\b/i] },
  { code: "A.5.15", patterns: [/\b(control de acceso|access control)\b/i] },
  { code: "A.5.16", patterns: [/\b(identidad|identity|usuario(s)? duplicad|sso)\b/i] },
  { code: "A.5.18", patterns: [/\b(permisos|rbac|derechos de acceso)\b/i] },
  { code: "A.8.20", patterns: [/\b(firewall|puerto(s)?|exposici[oó]n de red)\b/i] },
];

const BLOCKING_SEVERITIES = new Set(["medium", "high", "critical"]);
const OPEN_STATUS = "open";

export function findingToControls(finding: FindingLike): string[] {
  if (finding.status !== OPEN_STATUS) return [];
  if (!BLOCKING_SEVERITIES.has(finding.severity)) return [];
  const codes: string[] = [];
  for (const rule of RULES) {
    if (rule.patterns.some((p) => p.test(finding.title))) codes.push(rule.code);
  }
  return codes;
}

export function computeControlState(
  auto: AutoSignal,
  attestation: AttestationDTO | null
): { status: ControlStatus; refs: string[] } {
  if (attestation?.status === "not_applicable") return { status: "not_applicable", refs: [] };
  if (auto?.status === "non_compliant") return { status: "non_compliant", refs: auto.refs };
  if (attestation?.status === "non_compliant") return { status: "non_compliant", refs: [] };
  if (auto?.status === "compliant") return { status: "compliant", refs: [] };
  if (attestation?.status === "compliant") return { status: "declared", refs: [] };
  return { status: "no_signal", refs: [] };
}

type StatusKey = ControlStatus;

export interface Scorecard {
  frameworkId: string;
  percent: number;
  verdict: string;
  totals: Record<ThemeKind, Partial<Record<StatusKey, number>>>;
  controls: Array<{
    controlCode: string;
    theme: ThemeKind;
    titleEs: string;
    titleEn: string;
    status: ControlStatus;
    findingRefs: string[];
  }>;
}

const THEME_ORDER: ThemeKind[] = ["a5-organizational", "a6-people", "a7-physical", "a8-technological"];

export function buildScorecard(
  controls: ControlDefinition[],
  ctx: EvaluationContext,
  attestations: AttestationDTO[]
): Scorecard {
  const attByCode = new Map(attestations.map((a) => [a.controlCode, a]));

  // señal auto: findings abiertos mapeados → non_compliant;
  // remediation verified con vulnerabilityTitle == titulo de un finding mapeado → compliant SIEMPRE que ese control no tenga findings abiertos.
  const mappedFindings = ctx.openFindings
    .map((f) => ({ id: f.id, codes: findingToControls(f) }))
    .filter((m) => m.codes.length > 0);

  const openByControl = new Map<string, string[]>();
  for (const m of mappedFindings) {
    for (const code of m.codes) {
      openByControl.set(code, [...(openByControl.get(code) ?? []), m.id]);
    }
  }

  const verifiedByControl = new Set<string>();
  const openTitles = new Set(ctx.openFindings.map((f) => f.title.toLowerCase()));
  for (const r of ctx.verifiedRemediations) {
    const title = (r.vulnerabilityTitle ?? r.title)?.toLowerCase();
    if (title && openTitles.has(title)) continue; // hay finding abierto equivalente → la remediación no confirma
    // el remedio verificado solo aporta señal a un control si existe un finding mapeado con ese titulo (cerrado o el propio)
    // implementación MVP: matching por título contra el mapa de reglas del catálogo
    for (const rule of RULES) {
      if (rule.patterns.some((p) => p.test(title ?? ""))) verifiedByControl.add(rule.code);
    }
  }

  const results: Scorecard["controls"] = [];
  for (const control of controls) {
    const openRefs = openByControl.get(control.code) ?? [];
    let auto: AutoSignal = null;
    if (openRefs.length > 0) auto = { status: "non_compliant", refs: openRefs };
    else if (verifiedByControl.has(control.code)) auto = { status: "compliant", refs: [] };
    const state = computeControlState(auto, attByCode.get(control.code) ?? null);
    results.push({
      controlCode: control.code,
      theme: control.theme,
      titleEs: control.titleEs,
      titleEn: control.titleEn,
      status: state.status,
      findingRefs: state.refs,
    });
  }

  const totals = Object.fromEntries(
    THEME_ORDER.map((t) => [t, {} as Partial<Record<StatusKey, number>>])
  ) as Record<ThemeKind, Partial<Record<StatusKey, number>>>;

  let compliant = 0;
  let nonCompliant = 0;
  let noSignal = 0;
  for (const r of results) {
    totals[r.theme][r.status] = (totals[r.theme][r.status] ?? 0) + 1;
    if (r.status === "compliant") compliant += 1;
    else if (r.status === "non_compliant") nonCompliant += 1;
    else if (r.status === "no_signal") noSignal += 1;
  }

  const denominator = compliant + nonCompliant + noSignal;
  const percent = denominator === 0 ? 0 : Math.round((compliant / denominator) * 100);

  return {
    frameworkId: ISO27001_ANNEX_A.id,
    percent,
    verdict: verdictFromScore(percent),
    totals,
    controls: results,
  };
}
```

```ts
import { NextRequest, NextResponse } from "next/server";
import { and, eq } from "drizzle-orm";
import { z } from "zod";
import { createClient } from "@/shared/lib/supabase/server";
import { withRLS } from "@/shared/db/rls";
import { users } from "@/shared/db/schemas";
import { complianceAttestations } from "@/shared/db/schemas/compliance";
import { ISO27001_ANNEX_A } from "@/server/compliance/catalog/iso27001-annex-a";

export const dynamic = "force-dynamic";

const BODY = z
  .object({
    projectId: z.string().uuid(),
    controlCode: z.string().regex(/^A\.(5|6|7|8)\.\d{1,2}$/),
    status: z.enum(["compliant", "non_compliant", "not_applicable"]),
    note: z.string().min(1).max(4000).nullish(),
    evidenceUrl: z.string().url().https().nullish(),
  })
  .superRefine((v, ctx) => {
    if (v.status !== "compliant" && !v.note?.trim()) {
      ctx.addIssue({
        code: z.ZodIssueCode.custom,
        path: ["note"],
        message: "note requerida para non_compliant / not_applicable",
      });
    }
  });

export async function POST(request: NextRequest) {
  const supabase = createClient();
  const { data: auth } = await supabase.auth.getUser();
  const user = auth?.user;
  if (!user) return NextResponse.json({ error: "unauthorized" }, { status: 401 });

  const parsed = BODY.safeParse(await request.json());
  if (!parsed.success) return NextResponse.json({ error: "bad_request" }, { status: 400 });
  const { projectId, controlCode, status, note, evidenceUrl } = parsed.data;

  if (!ISO27001_ANNEX_A.controls.some((c) => c.code === controlCode)) {
    return NextResponse.json({ error: "control_invalido" }, { status: 400 });
  }

  return withRLS(user.id, async (tx) => {
    const [actor] = await tx
      .select({ role: users.role })
      .from(users)
      .where(eq(users.id, user.id))
      .limit(1);
    if (!actor || actor.role !== "admin") {
      return NextResponse.json({ error: "forbidden" }, { status: 403 });
    }

    const values = {
      projectId,
      frameworkId: ISO27001_ANNEX_A.id,
      controlCode,
      status,
      note: note ?? null,
      evidenceUrl: evidenceUrl ?? null,
      assertedBy: user.id,
      updatedAt: new Date(),
    };

    const [saved] = await tx
      .insert(complianceAttestations)
      .values(values)
      .onConflictDoUpdate({
        target: [
          complianceAttestations.projectId,
          complianceAttestations.frameworkId,
          complianceAttestations.controlCode,
        ],
        set: {
          status: values.status,
          note: values.note,
          evidenceUrl: values.evidenceUrl,
          assertedBy: values.assertedBy,
          updatedAt: values.updatedAt,
        },
      })
      .returning({ id: complianceAttestations.id, status: complianceAttestations.status });

    return NextResponse.json({ success: true, attestation: saved });
  });
}
```

```ts
  if (!project) return null;

  const [findings, remediations, attestations] = await Promise.all([
    tx
      .select({
        id: intelligenceFindings.id,
        projectId: intelligenceFindings.projectId,
        severity: intelligenceFindings.severity,
        status: intelligenceFindings.status,
        title: intelligenceFindings.title,
        evidence: intelligenceFindings.evidence,
      })
      .from(intelligenceFindings)
      .where(eq(intelligenceFindings.projectId, projectId))
      .limit(1000),
    tx
      .select({
        status: remediationActions.status,
        title: remediationActions.title,
        vulnerabilityTitle: remediationActions.vulnerabilityTitle,
        assessmentId: remediationActions.assessmentId,
      })
      .from(remediationActions)
      .where(eq(remediationActions.projectId, projectId))
      .limit(1000),
    tx
      .select()
      .from(complianceAttestations)
      .where(
        and(
          eq(complianceAttestations.projectId, projectId),
          eq(complianceAttestations.frameworkId, ISO27001_ANNEX_A.id)
        )
      ),
  ]);

  return buildScorecard(
    ISO27001_ANNEX_A.controls,
    {
      openFindings: findings.filter((f) => f.status === "open") as FindingLike[],
      verifiedRemediations: remediations.filter((r) => r.status === "verified"),
    },
    attestations.map((a) => ({
      controlCode: a.controlCode,
      status: a.status,
      note: a.note,
      evidenceUrl: a.evidenceUrl,
    }))
  );
}
```

```ts
import { NextRequest, NextResponse } from "next/server";
import { createHash } from "crypto";
import { createClient } from "@/shared/lib/supabase/server";
import { withRLS } from "@/shared/db/rls";
import { loadScorecard } from "@/server/compliance/service";

export const dynamic = "force-dynamic";

export async function GET(request: NextRequest) {
  const supabase = createClient();
  const { data: auth } = await supabase.auth.getUser();
  const user = auth?.user;
  if (!user) return NextResponse.json({ error: "unauthorized" }, { status: 401 });

  const projectId = new URL(request.url).searchParams.get("projectId");
  if (!projectId) return NextResponse.json({ error: "projectId requerido" }, { status: 400 });

  return withRLS(user.id, async (tx) => {
    const scorecard = await loadScorecard(tx, user.id, projectId);
    if (!scorecard) return NextResponse.json({ error: "not_found" }, { status: 404 });

    const pack = {
      frameworkId: scorecard.frameworkId,
      generatedAt: new Date().toISOString(),
      disclaimer: "Autoevaluación assistida por SCAUDIT; no constituye certificación ISO 27001.",
      scorecard,
    };
    const packJson = JSON.stringify(pack, null, 2);
    const sha256 = createHash("sha256").update(packJson).digest("hex");

    return new NextResponse(packJson, {
      status: 200,
      headers: {
        "Content-Type": "application/json; charset=utf-8",
        "Content-Disposition": `attachment; filename="iso27001-evidence-pack-${projectId.slice(0, 8)}.json"`,
        "X-SHA256": sha256,
      },
    });
  });
}
```

```tsx
"use client";

import { useEffect, useMemo, useState } from "react";
import { useTranslations } from "next-intl";

type ControlStatus = "compliant" | "non_compliant" | "not_applicable" | "declared" | "no_signal";
type ThemeKind = "a5" | "a6" | "a7" | "a8";

interface ControlRow {
  controlCode: string;
  theme: string;
  titleEs: string;
  titleEn: string;
  status: ControlStatus;
  findingRefs: string[];
}

interface ScorecardResponse {
  scorecard: {
    percent: number;
    verdict: string;
    controls: ControlRow[];
  };
}

const THEME_MAP: Record<string, ThemeKind> = {
  "a5-organizational": "a5",
  "a6-people": "a6",
  "a7-physical": "a7",
  "a8-technological": "a8",
};

export function ComplianceTab({ projectId }: { projectId: string }) {
  const t = useTranslations("compliance");
  const [data, setData] = useState<ScorecardResponse["scorecard"] | null>(null);
  const [error, setError] = useState<string | null>(null);
  const [filter, setFilter] = useState<"all" | "compliant" | "non_compliant" | "no_signal">("all");

  useEffect(() => {
    let alive = true;
    fetch(`/api/compliance/iso27001/overview?projectId=${projectId}`)
      .then(async (res) => {
        if (!res.ok) throw new Error(String(res.status));
        const body = (await res.json()) as ScorecardResponse;
        if (alive) setData(body.scorecard);
      })
      .catch((e: unknown) => { if (alive) setError(String((e as Error).message)); });
    return () => { alive = false; };
  }, [projectId]);

  const rows = useMemo(() => {
    if (!data) return [];
    const visible =
      filter === "all"
        ? data.controls
        : data.controls.filter((c) =>
            filter === "compliant" ? c.status === "compliant" || c.status === "declared" : c.status === filter
          );
    return visible;
  }, [data, filter]);

  if (error) return <p className="p-4 text-sm text-red-500">{error}</p>;
  if (!data) return <p className="p-4 text-sm opacity-70">…</p>;

  const chips: Record<ControlStatus, string> = {

    declared: "bg-sky-700/30 text-sky-300",

      </header>
      <p className="text-xs opacity-70">{t("pendingNotice")}</p>

      <div className="flex gap-2 text-xs">

          </button>
        ))}

        <tbody>

              <td className="py-2 pr-2 font-mono text-xs">{row.controlCode}</td>
```
